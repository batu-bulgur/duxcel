function inspectorSQLWheelTransport(amount,scale,surface,delivered){
 const requested=amount*scale*surface.height/surface.clientHeight*surface.dpr,rounded=Math.fround(requested),expectedDOM=rounded/surface.dpr,recovered=Number.isFinite(delivered?.deltaY)?delivered.deltaY*surface.clientHeight/surface.height/scale:null;
 const step=Math.max(2**-149,2**(Math.floor(Math.log2(Math.abs(requested)))-23)),halfULP=step/2,unitRoundoff=Number.EPSILON/2,gamma8=8*unitRoundoff/(1-8*unitRoundoff),bound=halfULP/surface.dpr*surface.clientHeight/surface.height/scale+gamma8*Math.max(1,Math.abs(amount));
 return{requested,rounded,expectedDOM,recovered,float32HalfULP:halfULP,uiErrorBound:bound,uiError:recovered===null?null:Math.abs(recovered-amount),model:'Pinned Chrome150 CDP double→float WebMouseWheelEvent delta; unzoomed top-level root frame; DOM divides by devicePixelRatio. Actual DOM delta must equal the rounded prediction; its unavoidable cast error is bounded separately on UI-unit recovery.',sources:['content/browser/devtools/protocol/input_handler.cc','third_party/blink/public/common/input/web_mouse_wheel_event.h','third_party/blink/common/input/web_mouse_wheel_event.cc','third_party/blink/renderer/core/events/wheel_event.cc'].map(file=>'https://raw.githubusercontent.com/chromium/chromium/5b586c06e0d27582900f17e2d59c5370d8d6e0bb/'+file)};
}
